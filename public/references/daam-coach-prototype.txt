                            <p class="text-lg text-gray-600">Understanding the rules, committees, and the accountability system.</p>

                        <div class="prose prose-green max-w-none text-gray-700 space-y-6">
                            <h3 class="text-2xl font-bold text-gray-900">What is DAAM?</h3>
                            <p>The Department of Activity Approval and Monitoring (DAAM) is an executive department under the USG Office of the Executive Secretary. It ensures all activities align with USG principles and the Lasallian Identity. <strong>Work Hours:</strong> Mondays to Saturdays, 8:00 AM to 10:00 PM.</p>

                            <h3 class="text-2xl font-bold text-gray-900 mt-8">The Six Committees</h3>
                            <ul class="space-y-3 list-disc pl-5">
                                <li><strong>APS (Activity Processing & Screening):</strong> Reviews and approves pre-activity requirements.</li>
                                <li><strong>SSC (Special Slips and Clearances):</strong> Handles Special Approval Slips (SAS) and Term-End Clearances.</li>
                                <li><strong>MST (MOA Screening Team):</strong> Reviews Memorandums of Agreement for internal and external partnerships.</li>
                                <li><strong>P&M (Publicity & Marketing):</strong> Approves publicity materials and monitors posting guidelines.</li>
                                <li><strong>AMT (Activity Monitoring Team):</strong> Evaluates the actual execution of events and participant feedback.</li>
                                <li><strong>ADM (Activity Documentation & Management):</strong> Approves post-activity requirements.</li>
                            </ul>

                            <h3 class="text-2xl font-bold text-gray-900 mt-8">The Penalty System (Demerits)</h3>
                            <p>DAAM enforces strict accountability. Violations result in demerit points ranging from 0.5 to over 14 points. Accumulating points leads to activity bans.</p>
                            <div class="bg-red-50 border-l-4 border-red-500 p-4 rounded-r-lg my-4">
                                <h4 class="font-bold text-red-800 mb-2">Common Offenses:</h4>
                                <ul class="list-disc pl-5 text-sm text-red-700 space-y-1">
                                    <li>Late submission of Pre-Acts or Post-Acts.</li>
                                    <li>Posting publication materials without DAAM P&M approval.</li>
                                    <li>Conducting an activity without approved Pre-Acts.</li>
                                    <li>Bypassing DAAM or SLIFE for signatures.</li>
                                </ul>
                            </div>
                        </div>

                window.scrollTo({ top: 0, behavior: 'smooth' });

        const systemInstruction = `Act as the USG DAAM Proposal Coach. I am a DLSU student leader planning an event. Guide me through drafting my Project Proposal Report (PPR) based on the Operations Manual.

1. Ask me questions ONE PHASE AT A TIME.

Phase 5: Output & Reminders. Provide the final clean text for me to copy. Remind me that I must manually handle the "Breakdown of Expenses" using the exact format "Php 0.00" and secure physical signatures.`;

        // Initialize chat history with the system instruction and the first assistant message (that is hardcoded in the HTML)
        let chatHistory = [
            {
                role: "user",
                parts: [{ text: "Hello! Let's draft my proposal." }]
            },
            {
                role: "model",
                parts: [{ text: "Hello! I'm the USG DAAM Proposal Coach. I'm here to help you draft your Project Proposal Report (PPR) based on the Operations Manual.\n\nTo get started, we are entering **Phase 1: Discovery**.\n\nWhat is the title of your event, who is your target audience, and what are the proposed dates?" }]
            }
        ];

        const chatWindow = document.getElementById('chat-window');
        const chatInput = document.getElementById('chat-input');
        const sendButton = document.getElementById('send-button');
        
        let isWaitingForResponse = false;

        // Handle Enter key to send
        chatInput.addEventListener('keypress', function (e) {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                if (!isWaitingForResponse) {
                    handleSendMessage();
                }
            }
        });

        async function handleSendMessage() {
            if (isWaitingForResponse) return;

            const message = chatInput.value.trim();
            if (!message) return;

            // Lock UI while processing
            isWaitingForResponse = true;
            chatInput.disabled = true;
            sendButton.disabled = true;
            sendButton.classList.add('opacity-50', 'cursor-not-allowed');

            // 1. Add User Message to UI
            appendMessage(message, 'user');
            chatInput.value = '';
            
            // 2. Add User Message to History
            chatHistory.push({ role: "user", parts: [{ text: message }] });

            // 3. Show Loading Indicator
            const loadingId = appendLoading();

            // 4. Call Gemini API
            try {
                const responseText = await fetchGeminiResponse(chatHistory);
                
                // 5. Remove Loading Indicator
                document.getElementById(loadingId).remove();

                // 6. Add AI Response to UI
                appendMessage(responseText, 'model');

                // 7. Add AI Response to History
                chatHistory.push({ role: "model", parts: [{ text: responseText }] });

            } catch (error) {
                console.error("Detailed API Error:", error);
                document.getElementById(loadingId).remove();
                
                let errorMessage = "Sorry, I encountered an error connecting to the AI. ";
                
                // Provide actionable feedback based on the error
                if (error.message.includes("API_KEY_INVALID") || error.message.includes("API key")) {
                    errorMessage += "<strong>Error: Missing/Invalid API Key.</strong> If you exported this file to run locally, you must paste your own Google Gemini API key into the 'apiKey' variable in the code.";
                } else {
                    errorMessage += `<br/><span class="text-xs text-red-500">System Log: ${error.message}</span>`;
                }

                appendMessage(errorMessage, 'model', true);
                
                // Remove the last user message from history so they can retry seamlessly
                chatHistory.pop(); 
            } finally {
                // Unlock UI
                isWaitingForResponse = false;
                chatInput.disabled = false;
                sendButton.disabled = false;
                sendButton.classList.remove('opacity-50', 'cursor-not-allowed');
                chatInput.focus();
            }
        }

        async function fetchGeminiResponse(history) {
            // NOTE: If you are running this outside of this Canvas environment, 
            // you must place your Google Gemini API key inside the quotes below.
            const apiKey = ""; 
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

            const payload = {
                contents: history,
                systemInstruction: {
                    parts: [{ text: systemInstruction }]
                }
            };

            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            const result = await response.json();
            
            if (!response.ok) {
                console.error("API Error Payload:", result);
                throw new Error(result.error?.message || `HTTP error! status: ${response.status}`);
            }
            
            if (result.candidates && result.candidates[0] && result.candidates[0].content && result.candidates[0].content.parts) {
                return result.candidates[0].content.parts[0].text;
            } else {
                console.error("Unexpected API Response Structure:", result);
                throw new Error("Invalid API response structure received from the server.");
            }
        }

        function appendMessage(text, sender, isError = false) {
            const msgDiv = document.createElement('div');
            msgDiv.className = `flex items-start max-w-[80%] ${sender === 'user' ? 'ml-auto flex-row-reverse' : ''}`;
            
            // Format text (simple markdown bold processing for UI)
            const formattedText = text.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>').replace(/\n/g, '<br/>');

            if (sender === 'user') {
                msgDiv.innerHTML = `
                    <div class="bg-green-600 p-3 px-5 rounded-2xl rounded-tr-none shadow-sm text-white">
                        <p>${formattedText}</p>
                    </div>
                `;
            } else {
                msgDiv.innerHTML = `
                    <div class="w-8 h-8 bg-green-100 rounded-full flex items-center justify-center ${sender === 'user' ? 'ml-2' : 'mr-2'} flex-shrink-0 mt-1">
                        <i class="fa-solid fa-robot text-sm text-green-600"></i>
                    </div>
                    <div class="bg-white p-4 rounded-2xl rounded-tl-none shadow-sm border ${isError ? 'border-red-300 text-red-600' : 'border-gray-100 text-gray-700'}">
                        <p>${formattedText}</p>
                    </div>
                `;
            }

            chatWindow.appendChild(msgDiv);
            scrollToBottom();
        }

        function appendLoading() {
            const id = 'loading-' + Date.now();
            const msgDiv = document.createElement('div');
            msgDiv.id = id;
            msgDiv.className = `flex items-start max-w-[80%]`;
            
            msgDiv.innerHTML = `
                <div class="w-8 h-8 bg-green-100 rounded-full flex items-center justify-center mr-2 flex-shrink-0 mt-1">
                    <i class="fa-solid fa-robot text-sm text-green-600"></i>
                </div>
                <div class="bg-white p-4 rounded-2xl rounded-tl-none shadow-sm border border-gray-100 text-gray-700 flex space-x-1 items-center h-12">
                    <div class="w-2 h-2 bg-gray-400 rounded-full animate-bounce"></div>
                    <div class="w-2 h-2 bg-gray-400 rounded-full animate-bounce" style="animation-delay: 0.2s"></div>
                    <div class="w-2 h-2 bg-gray-400 rounded-full animate-bounce" style="animation-delay: 0.4s"></div>
                </div>
            `;

            chatWindow.appendChild(msgDiv);
            scrollToBottom();
            return id;
        }

        function scrollToBottom() {
            chatWindow.scrollTop = chatWindow.scrollHeight;
        }
